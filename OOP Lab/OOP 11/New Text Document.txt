#include <iostream>
#include <vector>
#include <memory>  // Required for smart memory management
#include <stdexcept>
#include <string>

using namespace std;

class Book {
protected: // Changed to protected so derived classes can read them cleanly if needed
    string title;
    int id;
    int quantity;

public:
    Book(string s, int i, int q) : title(s), id(i), quantity(q) {}
    virtual ~Book() = default; // Essential for virtual tables and safe deletion via base pointer

    // Scenario A: Using a Function Template for polymorphic searching
    template <typename T>
    void search(const T& query) {
        // If T is int, we check ID. If T is string, we check title.
        // We use compile-time conditional checking (C++17 if constexpr)
        if constexpr (is_same_v<T, int>) {
            if (id == query) display();
            else cout << "Oops! Not Found by ID: " << query << endl;
        } 
        else if constexpr (is_same_v<T, string> || is_same_v<T, const char*>) {
            if (title == query) display();
            else cout << "Oops! Not Found by Title: " << query << endl;
        }
    }

    // Operator Overloading
    void operator+(int x) { quantity += x; }
    void operator-(int x) { 
        if (quantity >= x) quantity -= x; 
        else throw runtime_error("Underflow: Not enough copies available.");
    }

    int getId() const { return id; }
    int getQuantity() const { return quantity; }

    virtual void display() const {
        cout << "Book Title : " << title << "\n"
             << "Book ID    : " << id << "\n"
             << "Available  : " << quantity << endl;
    }
};

// Fixed to PUBLIC inheritance to maintain the "is-a" structural contract
class Ebook : public Book {
    double fileSize;

public:
    Ebook(string s, int i, int q, double f) : Book(s, i, q), fileSize(f) {}
    
    void display() const override {
        Book::display();
        cout << "File size  : " << fileSize << " MB" << endl;
    }
};

// Scenario B: Class Template turning Library into a generic resource controller
template <typename ResourceType>
class Library {
private:
    // Storing unique_ptr prevents object slicing and allows pure runtime polymorphism
    vector<unique_ptr<ResourceType>> inventory;

public:
    void addResource(unique_ptr<ResourceType> res) {
        inventory.push_back(move(res)); // Transferring ownership into the container
    }

    void displayInventory() const {
        for (const auto& item : inventory) {
            item->display(); // Resolves dynamically at runtime to Book::display or Ebook::display
            cout << "---------------------\n";
        }
    }

    void borrowResource(int id) {
        for (auto& item : inventory) {
            if (id == item->getId()) {
                if (item->getQuantity() <= 0) {
                    throw runtime_error("Resource currently out of stock.");
                }
                *item - 1; // Invokes operator-
                cout << "Resource successfully borrowed!\n";
                return;
            }
        }
        throw runtime_error("Resource ID not found in system registers.");
    }
};

int main() {
    // Instantiate our class template specifically for Book resources
    Library<Book> engineeringLibrary;

    // Use make_unique to instantiate elements safely on the heap
    engineeringLibrary.addResource(make_unique<Book>("C++ Primer", 101, 5));
    engineeringLibrary.addResource(make_unique<Ebook>("Advanced OS", 102, 3, 14.2));

    cout << "=== University Inventory ===\n";
    engineeringLibrary.displayInventory();

    // Testing our Function Template
    cout << "=== Compile-Time Template Search ===\n";
    Book physicalBook("Design Patterns", 301, 2);
    physicalBook.search(301);       // Resolves automatically to the int branch
    physicalBook.search(string("Design Patterns")); // Resolves to the string branch

    cout << "\n=== Transaction Execution ===\n";
    try {
        engineeringLibrary.borrowResource(102); // Works perfectly on Ebook now without slicing!
    } catch (const exception& e) {
        cout << "System Exception Caught: " << e.what() << endl;
    }

    return 0;
}
